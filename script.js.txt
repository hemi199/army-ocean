const wheel = document.getElementById("wheelBox");
const spinButton = document.getElementById("spinBtn");
const scoreElement = document.getElementById("score");
const resultElement = document.getElementById("result");

const rewards = [10, 20, 30, 40, 50, 60, 70];

let totalScore = 0;
let currentRotation = 0;
let isSpinning = false;

spinButton.addEventListener("click", function () {
  if (isSpinning) {
    return;
  }

  isSpinning = true;
  spinButton.disabled = true;
  resultElement.textContent = "";

  const randomIndex = Math.floor(Math.random() * rewards.length);
  const sliceAngle = 360 / rewards.length;

  const extraSpins = 6 + Math.floor(Math.random() * 3);
  const targetAngle =
    360 - (randomIndex * sliceAngle + sliceAngle / 2);

  const smallRandomMove =
    Math.floor(Math.random() * 20) - 10;

  currentRotation +=
    extraSpins * 360 + targetAngle + smallRandomMove;

  wheel.style.transform = `rotate(${currentRotation}deg)`;

  setTimeout(function () {
    const earnedPoints = rewards[randomIndex];

    totalScore += earnedPoints;
    scoreElement.textContent = totalScore;

    resultElement.textContent =
      `مبروك! حصلتِ على ${earnedPoints} نقطة 🎉`;

    spinButton.disabled = false;
    isSpinning = false;
  }, 4000);
});
