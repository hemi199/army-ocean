// العناصر الخاصة بعجلة النقاط
const wheel = document.getElementById('wheel');
const spinBtn = document.getElementById('spinBtn');
const scoreEl = document.getElementById('score');
const resultEl = document.getElementById('result');

const rewards = [10, 20, 30, 40, 50, 60, 70];
let currentRotation = 0;
let totalScore = 0;
let spinning = false;

// حدث الضغط على زر الدوران
if (spinBtn) {
  spinBtn.addEventListener('click', () => {
    if (spinning) return;
    
    spinning = true;
    spinBtn.disabled = true;
    resultEl.textContent = '';

    const extraSpins = 6 + Math.floor(Math.random() * 4);
    const prizeIndex = Math.floor(Math.random() * rewards.length);
    const sliceAngle = 360 / rewards.length;
    const targetAngle = 360 - (prizeIndex * sliceAngle + sliceAngle / 2);
    const randomJitter = Math.floor(Math.random() * 25) - 12;

    currentRotation += extraSpins * 360 + targetAngle + randomJitter;
    wheel.style.transform = `rotate(${currentRotation}deg)`;

    setTimeout(() => {
      const gained = rewards[prizeIndex];
      totalScore += gained;
      scoreEl.textContent = totalScore;
      resultEl.textContent = `🎉 ربحتي ${gained} نقطة!`;
      spinBtn.disabled = false;
      spinning = false;
    }, 4000);
  });
}
